CREATE VIEW PruebaV AS
SELECT productos.nombre, productos.idCategoria, productos.precio*productos.existencia AS valorinventario
FROM productos


CREATE VIEW PruebaV AS
SELECT productos.nombre, SUM(productos.precio) AS valorinventario
FROM productos

WHERE nombre="Paleta de chocolate"

CREATE VIEW vistaprovprod AS
SELECT proveedores.nombre, concat(productos.nombre, ", ", categoria.nombre) AS ProductoYCategoria, productos.existencia
FROM proveedores, productos, categoria, detalle_producto_proveedor
WHERE (detalle_producto_proveedor.idProducto=productos.idProducto) AND (detalle_producto_proveedor.idProveedor=proveedores.id) AND (productos.idCategoria=categoria.id)
