CREATE FUNCTION verdaderofalso(numero int)
	returns varchar(5) DETERMINISTIC
BEGIN
CASE
   	WHEN numero%2=0 THEN
   	RETURN 'par';
    ELSE RETURN 'impar';
END CASE;
END//
DELIMITER ;

CREATE FUNCTION mayor3(valor1 int, valor2 int, valor3 int)
RETURNS int DETERMINISTIC
BEGIN
	if valor1>valor2 THEN
    	if valor1>valor3 THEN
        	RETURN valor1;
        ELSE
        	RETURN valor3;
        END IF;
    ELSEIF valor2>valor3 THEN
    	RETURN valor2;
    END IF;
END//
DELIMITER ;

DELIMITER //
CREATE FUNCTION dias(num int)
RETUNRS int
BEGIN
	CASE
    	WHEN num=1 THEN RETURN 'lunes';
        WHEN num=2 THEN RETURN 'martes';
        WHEN num=3 THEN RETURN 'miercoles';
        WHEN num=4 THEN RETURN 'jueves';
        WHEN num=5 THEN RETURN 'viernes';
        WHEN num=6 THEN RETURN 'sabado';
        WHEN num=7 THEN RETURN 'domingo';
        ELSE RETURN 'no es un dia de la semana';
    END CASE;
END//
DELIMITER;

CREATE FUNCTION CalculaMontoInvertido(nombreprod varchar(45))
	RETURNS int DETERMINISTIC
BEGIN
	DECLARE total int;
	SELECT (precio*existencia) INTO total FROM productos WHERE nombre=nombreprod;
    SET total=total;
    RETURN total;
END//
DELIMITER ;
